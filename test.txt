</div>

<script>
    function fetchData() {
        fetch('https://abdalaziz.vercel.app/items/', {
            method: 'GET',
            headers: {
                'Content-Type': 'application/json'
            }
        })
            .then(response => response.json())
            .then(data => {
                const container = document.getElementById('items-container');
                container.innerHTML = '';  // إفراغ المحتوى قبل عرض البيانات

                data.forEach(item => {
                    const itemCard = document.createElement('div');
                    itemCard.classList.add('item-card');

                    const nameElement = document.createElement('h3');
                    nameElement.textContent = `اسم العنصر: ${item.name}`;

                    const descriptionElement = document.createElement('p');
                    descriptionElement.textContent = `المفتاح المشفر: ${item.description}`;  

                    const copyBtn = document.createElement('button');
                    copyBtn.classList.add('copy-btn');
                    copyBtn.textContent = 'نسخ المفتاح';
                    copyBtn.onclick = () => copyToClipboard(item.description); 

                    itemCard.appendChild(nameElement);
                    itemCard.appendChild(descriptionElement);
                    itemCard.appendChild(copyBtn);

                    container.appendChild(itemCard);
                });
            })
            .catch(error => console.error('خطأ في جلب البيانات:', error));
    }

    function copyToClipboard(text) {
        // نسخ النص المشفر إلى الحافظة
        const tempInput = document.createElement('input');
        document.body.appendChild(tempInput);
        tempInput.value = text;  
        tempInput.select();
        document.execCommand('copy');
        document.body.removeChild(tempInput);
        alert('تم نسخ المفتاح المشفر: ' + text);
    }
</script>
